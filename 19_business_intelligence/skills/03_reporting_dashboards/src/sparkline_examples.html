JavaScript Sparkline Generator

function generateSparkline(data, width = 100, height = 20) {
    const max = Math.max(...data);
    const min = Math.min(...data);
    const range = max - min || 1;
    
    const points = data.map((value, index) => {
        const x = (index / (data.length - 1)) * width;
        const y = height - ((value - min) / range) * height;
        return `${x},${y}`;
    }).join(' ');
    
    return `
        <svg width="${width}" height="${height}" viewBox="0 0 ${width} ${height}">
            <polyline 
                points="${points}"
                fill="none"
                stroke="#2171B5"
                stroke-width="1.5"
            />
        </svg>
    `;
}

// Usage:
const revenueData = [100, 110, 105, 120, 115, 130, 125, 140, 135, 150];
const sparkline = generateSparkline(revenueData);
