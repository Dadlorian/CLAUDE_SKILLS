# PRIMARY DOMAIN - Geolocation Routing
# ============================================================

# ============================================================
# TRAFFIC POLICIES (Advanced Routing)
# ============================================================

# Traffic Policy: Intelligent Routing
Policy Name: intelligent-geolocation-with-fallback
Version: 1

Rules:
  1. Client Location: North America
     Check Health: us-east-dc-health-check
     If Healthy: Return 203.0.113.10 (US East)
     If Unhealthy: Return 203.0.113.11 (US West Fallback)
     TTL: 60

  2. Client Location: Europe
     Check Health: eu-central-dc-health-check
     If Healthy: Return 192.0.2.20 (EU Central)
     If Unhealthy: Return 192.0.2.21 (EU Secondary)
     TTL: 60

  3. Client Location: Asia Pacific
     Check Health: apac-dc-health-check
     If Healthy: Return 198.51.100.30 (APAC Singapore)
     If Unhealthy: Return 198.51.100.31 (APAC Sydney)
     TTL: 60

  4. Default: Return Global Anycast IP
     TTL: 300
