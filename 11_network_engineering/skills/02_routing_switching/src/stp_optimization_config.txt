! Spanning Tree Protocol Optimization Configuration
! RSTP with multiple VLANs and optimization

hostname Switch-A-Primary-Root
!
! Enable RSTP (802.1w)
spanning-tree mode rapid-pvst
!
! ***** Root Bridge Configuration *****
! Set as primary root for all VLANs
spanning-tree vlan 1,10,20,30,40 root primary
!
! ***** Port Configuration *****
!
! Backbone/Trunk Ports
interface GigabitEthernet0/47
  description Trunk to Switch-B
  spanning-tree vlan 1,10,20,30,40 cost 2000
  spanning-tree vlan 1,10,20,30,40 port-priority 128
  spanning-tree guard root      ! Root guard prevents rogue root
  spanning-tree link-type point-to-point
  no shutdown
!
interface GigabitEthernet0/48
  description Trunk to Switch-C
  spanning-tree vlan 1,10,20,30,40 cost 4000
  spanning-tree vlan 1,10,20,30,40 port-priority 192
  spanning-tree guard root
  spanning-tree link-type point-to-point
  no shutdown
!
! Access Ports with PortFast
interface range GigabitEthernet0/1-46
  spanning-tree portfast
  spanning-tree bpduguard enable
  spanning-tree cost 100
!
! RSTP Features
spanning-tree pathcost method long    ! Use modern costs (802.1w)
spanning-tree vlan 1,10,20,30,40 priority 4096
spanning-tree vlan 1,10,20,30,40 hello-time 2
spanning-tree vlan 1,10,20,30,40 forward-time 15
spanning-tree vlan 1,10,20,30,40 max-age 20
!
! Loop Guard and BPDU Guard
spanning-tree loopguard default
spanning-tree portfast default
spanning-tree portfast bpduguard default
!
! Monitoring
spanning-tree bpdu rate-limit
show spanning-tree
