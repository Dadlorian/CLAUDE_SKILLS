! OSPF Single Area Configuration - Router-A
! Campus Network with Single Area 0

hostname Router-A
!
! Loopback interface for Router ID
interface Loopback 0
  ip address 192.168.1.1 255.255.255.255
!
! LAN Interface - Building A
interface GigabitEthernet0/0
  description Connection to Building-A
  ip address 10.1.0.1 255.255.255.0
  ip ospf cost 1
  no shutdown
!
! WAN Interface - Building B
interface GigabitEthernet0/1
  description Connection to Building-B via WAN
  bandwidth 10000
  ip address 10.2.0.1 255.255.255.0
  ip ospf cost 100
  no shutdown
!
! OSPF Configuration
router ospf 1
  router-id 192.168.1.1

  ! Network statements with area
  network 10.1.0.0 0.0.0.255 area 0
  network 10.2.0.0 0.0.0.255 area 0
  network 192.168.1.0 0.0.0.255 area 0

  ! Passive interfaces (no hellos)
  passive-interface default
  no passive-interface GigabitEthernet0/0
  no passive-interface GigabitEthernet0/1

  ! Authentication - MD5
  area 0 authentication message-digest

  ! Tuning
  auto-cost reference-bandwidth 10000
  timers throttle spf 50 200 5000

  ! Logging
  log-adjacency-changes
!
! Interface-level authentication
interface GigabitEthernet0/0
  ip ospf message-digest-key 1 md5 OSPF_PASSWORD_101
!
interface GigabitEthernet0/1
  ip ospf message-digest-key 1 md5 OSPF_PASSWORD_101
!
! Verification commands:
! show ip ospf neighbor
! show ip ospf interface
! show ip route ospf
! show ip ospf database
