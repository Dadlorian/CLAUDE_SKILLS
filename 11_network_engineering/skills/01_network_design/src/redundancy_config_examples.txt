! Multiple approaches for network redundancy

! ========================================
! EXAMPLE 1: HSRP Gateway Redundancy
! ========================================

! Distribution-A (Primary)
router ospf 1
  !

interface Vlan 20
  description Finance VLAN
  ip address 10.0.20.2 255.255.255.0
  standby 20 ip 10.0.20.1
  standby 20 priority 150
  standby 20 preempt
  standby 20 timers 3 10
  standby 20 authentication md5 key-string hsrpkey123
  !

! Distribution-B (Secondary)
interface Vlan 20
  description Finance VLAN
  ip address 10.0.20.3 255.255.255.0
  standby 20 ip 10.0.20.1
  standby 20 priority 100
  standby 20 preempt
  standby 20 timers 3 10
  standby 20 authentication md5 key-string hsrpkey123
  !

! Verification
! show standby brief
! show standby vlan 20
! Active router will show "Active" status
! Standby router will show "Standby" status

! EXAMPLE 2: VRRP Gateway Redundancy
! ========================================

! Arista Example
interface Vlan 30
  description Engineering VLAN
  ip address 10.0.30.2 255.255.255.0
  ip virtual-router address 10.0.30.1
  ip virtual-router mac-address 00:00:5e:00:01:1e
  vrrp 30 priority 100
  vrrp 30 advertisement-interval 1000
  vrrp 30 preempt
  !

! ========================================
! EXAMPLE 3: EtherChannel Load Balancing
! ========================================

! Access switch uplink configuration
interface Port-channel 10
  switchport mode trunk
  switchport trunk allowed vlan 1-100
  spanning-tree portfast disable
  !

interface Gi0/0/47
  description Uplink to Distribution-A
  channel-group 10 mode active
  !

interface Gi0/0/48
  description Uplink to Distribution-B
  channel-group 10 mode active
  !

! Verify load balancing
! show etherchannel summary
! show etherchannel port-channel

! ========================================
! EXAMPLE 4: Dual ISP Failover with BGP
! ========================================

! Core Router BGP configuration

router bgp 65001
  bgp router-id 10.255.3.1
  !
  neighbor 198.51.100.1 remote-as 65100
  neighbor 198.51.100.1 description ISP-A
  neighbor 203.0.113.1 remote-as 65101
  neighbor 203.0.113.1 description ISP-B
  !
  address-family ipv4
    neighbor 198.51.100.1 activate
    neighbor 203.0.113.1 activate
    !
    ! Primary ISP via local-preference
    neighbor 198.51.100.1 route-map ISP_A_IN in
    neighbor 203.0.113.1 route-map ISP_B_IN in
    !
    default-information originate
    !
  exit-address-family
  !

! Route maps to set local preference
route-map ISP_A_IN permit 10
  set local-preference 200
  !

route-map ISP_B_IN permit 10
  set local-preference 100
  !

! BFD for faster failover detection
interface Gi0/0/4
  description Link to ISP-A
  ip address 198.51.100.2 255.255.255.0
  bfd interval 300 min_rx 300 multiplier 3
  !

interface Gi0/0/5
  description Link to ISP-B
  ip address 203.0.113.2 255.255.255.0
  bfd interval 300 min_rx 300 multiplier 3
  !

! BGP timers for BFD
router bgp 65001
  neighbor 198.51.100.1 timers 3 10
  neighbor 203.0.113.1 timers 3 10
  !

! Failover behavior:
! Normal: 90% traffic ISP-A (LP=200), 10% ISP-B (LP=100)
! ISP-A fails: 100% traffic ISP-B (automatic via BGP)
! Convergence time: <3 seconds (or <1 sec with BFD)

! ========================================
! EXAMPLE 5: STP for Access Layer
! ========================================

! Distribution switch (Root Primary)
spanning-tree mode rapid-pvst
spanning-tree priority 8192  ! Lower is better
spanning-tree vlan 10-50,100,110,120 priority 8192

! Access switch (Root Secondary)
spanning-tree mode rapid-pvst
spanning-tree priority 16384  ! Higher than primary

! Access ports
interface range Gi0/0/1-24
  spanning-tree portfast
  spanning-tree bpdu-guard enable
  !

! Uplink ports
interface range Gi0/0/47-48
  spanning-tree cost 1000  ! Lower cost = preferred path
  !

! Verify spanning tree
! show spanning-tree vlan 10
! show spanning-tree summary
! Active topology: Access-A1 → Distribution-A (primary path)
! Backup topology: Access-A1 → Distribution-B (STP blocks)

! ========================================
! EXAMPLE 6: Core Layer Redundancy
! ========================================

! Core-1 configuration
router bgp 65001
  neighbor 10.254.0.1 remote-as 65001
  neighbor 10.254.0.1 description Core-2
  neighbor 10.254.1.1 remote-as 65001
  neighbor 10.254.1.1 description Distribution-A
  !
  address-family ipv4
    neighbor 10.254.0.1 activate
    neighbor 10.254.1.1 activate
    !
    maximum-paths 2
    !
  exit-address-family
  !

! BFD for core-to-core link
interface Gi0/0/1
  description Link to Core-2
  ip address 10.254.0.0 255.255.255.254
  bfd interval 300 min_rx 300 multiplier 3
  !

! Verify
! show ip bgp summary
! show ip route summary
! show interface status
! Failover: If Core-1 fails, all traffic via Core-2

! ========================================
! EXAMPLE 7: Monitoring Redundancy
! ========================================

! Track object for HSRP
track 100 ip sla 1 reachability
  delay down 10 up 1
  !

ip sla 1
  icmp-echo 10.0.20.254  ! Probe destination
  frequency 10
  !

interface Vlan 20
  standby 20 track 100 decrement 20
  ! If route to 10.0.20.254 fails, decrease priority by 20
  ! HSRP becomes active on standby router
  !

! ========================================
! EXAMPLE 8: Dual Firewall Configuration
! ========================================

! Core layer traffic to firewalls

interface Gi0/0/2
  description Link to Firewall-1
  ip address 10.254.10.0 255.255.255.254
  !

interface Gi0/0/3
  description Link to Firewall-2
  ip address 10.254.11.0 255.255.255.254
  !

! BGP to firewalls
router bgp 65001
  neighbor 10.254.10.1 remote-as 65002
  neighbor 10.254.10.1 description Firewall-1
  neighbor 10.254.11.1 remote-as 65002
  neighbor 10.254.11.1 description Firewall-2
  !
  address-family ipv4
    neighbor 10.254.10.1 activate
    neighbor 10.254.11.1 activate
    !
  exit-address-family
  !

! Firewall should also have HA configured
! Failover: If Firewall-1 fails, Firewall-2 handles traffic
