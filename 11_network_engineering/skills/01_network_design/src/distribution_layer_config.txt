! ==============================================
! UPLINK CONFIGURATION TO CORE
! ==============================================

  description Uplink to Core Switches
  no switchport
  ip address 10.254.1.1 255.255.255.254
  speed 100000
  !

interface Gi0/0/47
  description Uplink Member 1 to Core
  no switchport
  channel-group 1 mode active
  !

  description Uplink Member 2 to Core
  no switchport
  channel-group 1 mode active
  !

! ==============================================
! ACCESS LAYER UPLINK CONFIGURATION
! ==============================================

interface Port-channel 10
  description Access Layer Uplink
  switchport mode trunk
  switchport trunk allowed vlan 1-120,200
  spanning-tree portfast disable
  !

interface Gi0/0/1
  switchport mode trunk
  channel-group 10 mode active
  !

interface Gi0/0/2
  switchport mode trunk
  channel-group 10 mode active
  !

! ==============================================
! ACCESS CONTROL LISTS
! ==============================================

ip access-list extended FINANCE_ALLOW
  permit ip 10.0.20.0 0.0.0.255 10.0.120.0 0.0.0.255
  permit ip 10.0.20.0 0.0.0.255 10.255.0.0 0.0.0.255
  deny ip 10.0.20.0 0.0.0.255 10.0.10.0 0.0.0.255
  deny ip 10.0.20.0 0.0.0.255 10.0.30.0 0.0.0.255
  permit ip any any

! Apply ACL to VLAN
interface Vlan 20
  ip access-group FINANCE_ALLOW in
  !

! QoS CONFIGURATION
