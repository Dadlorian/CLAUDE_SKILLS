! =====================================
! Object Definitions
! =====================================

object network WEB_SERVERS
 description Web Server Pool
 subnet 10.1.10.0 255.255.255.0

object network APP_SERVERS
 description Application Server Pool
 subnet 10.1.20.0 255.255.255.0

object network DB_SERVERS
 description Database Server Pool
 subnet 10.1.30.0 255.255.255.0

object network INTERNAL_NETWORKS
 description All Internal Networks
 subnet 10.1.0.0 255.255.0.0

object network GUEST_USERS
 description Guest User Network
 subnet 192.168.100.0 255.255.255.0

object-group network TRUSTED_NETWORKS
 description All Trusted Internal Networks
 network-object 10.1.0.0 255.255.0.0
 network-object 172.16.0.0 255.240.0.0

object-group service WEB_SERVICES tcp
 description Standard Web Services
 port-object eq http
 port-object eq https
 port-object eq 8080
 port-object eq 8443

object-group service MANAGEMENT_SERVICES tcp
 description Administrative Services
 port-object eq ssh
 port-object eq 22
 port-object eq 3389
 port-object eq 5900

object-group service DB_SERVICES tcp
 description Database Services
 port-object eq 3306
 port-object eq 5432
 port-object eq 1433

! =====================================
! Access Control Policies
! =====================================

! Outside to DMZ - Web Services Only
access-list OUTSIDE_TO_DMZ extended permit tcp any object WEB_SERVERS object-group WEB_SERVICES log
access-list OUTSIDE_TO_DMZ extended deny ip any any log

! DMZ to Inside - Limited Access
access-list DMZ_TO_INSIDE extended permit tcp object WEB_SERVERS object APP_SERVERS object-group DB_SERVICES log
access-list DMZ_TO_INSIDE extended permit tcp object WEB_SERVERS object DB_SERVERS eq 3306 log
access-list DMZ_TO_INSIDE extended deny ip any any log

! Inside to Outside - Selective Outbound
access-list INSIDE_TO_OUTSIDE extended permit tcp object-group TRUSTED_NETWORKS any eq 80 log
access-list INSIDE_TO_OUTSIDE extended permit tcp object-group TRUSTED_NETWORKS any eq 443 log
access-list INSIDE_TO_OUTSIDE extended permit tcp object-group TRUSTED_NETWORKS any eq 53 log
access-list INSIDE_TO_OUTSIDE extended permit udp object-group TRUSTED_NETWORKS any eq 53 log
access-list INSIDE_TO_OUTSIDE extended permit udp object-group TRUSTED_NETWORKS any eq 123 log
access-list INSIDE_TO_OUTSIDE extended deny ip any any log

! Management Access - Restricted
access-list MANAGEMENT_IN extended permit tcp object-group TRUSTED_NETWORKS any object-group MANAGEMENT_SERVICES log
access-list MANAGEMENT_IN extended deny ip any any log

! Apply Access Lists
access-group OUTSIDE_TO_DMZ in interface outside
access-group DMZ_TO_INSIDE in interface dmz
access-group INSIDE_TO_OUTSIDE in interface inside
access-group MANAGEMENT_IN in interface management

nat (dmz,outside) static 203.0.113.10 10.1.10.5 netmask 255.255.255.255 service tcp 443 443

 group 15

 set transform-set SITE_TO_SITE
 set pfs group15
 set lifetime seconds 3600

crypto map VPN_CMAP interface outside

crypto ikev2 enable outside
crypto ikev2 dpd 10 3 on-demand
