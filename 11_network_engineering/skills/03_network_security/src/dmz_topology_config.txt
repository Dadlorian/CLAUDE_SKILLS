! DMZ Topology Configuration
! Multi-Tier Demilitarized Zone Design
! Production-Grade Enterprise Architecture

! =====================================
! Network Topology Overview
! =====================================

! Internet
!     |
! [Perimeter Firewall - Outside Interface]
!     |
! [DMZ Tier 1 - Web Layer - VLAN 10]
!   10.1.10.0/24
!   - Load Balancers (10.1.10.1)
!   - Web Servers (10.1.10.5-10)
!     |
! [Internal Firewall]
!     |
! [DMZ Tier 2 - App Layer - VLAN 20]
!   10.1.20.0/24
!   - App Servers (10.1.20.5-10)
!   - Cache Servers (10.1.20.30-35)
!     |
! [Database Firewall]
!     |
! [Data Tier - Internal - VLAN 30]
!   10.1.30.0/24
!   - Database Servers (10.1.30.5-10)
!   - Backup Systems (10.1.30.20-25)

! =====================================
! VLAN Configuration
! =====================================

vlan 10
 name DMZ_WEB_TIER
 description Web servers, load balancers, WAF

vlan 20
 name DMZ_APP_TIER
 description Application servers, middleware

vlan 30
 name DATA_TIER
 description Database servers, internal use

vlan 100
 name MANAGEMENT
 description Device management interface

! =====================================
! Router Configuration (Core Router)
! =====================================

interface Vlan10
 ip address 10.1.10.1 255.255.255.0
 description DMZ Web Tier Gateway

interface Vlan20
 ip address 10.1.20.1 255.255.255.0
 description DMZ App Tier Gateway

interface Vlan30
 ip address 10.1.30.1 255.255.255.0
 description Data Tier Gateway

! =====================================
! Firewall Configuration - Web Tier (DMZ 1)
! =====================================

! Inbound from Internet
access-list WEB_TIER_INBOUND extended permit tcp any 10.1.10.0 255.255.255.0 eq 80 log
access-list WEB_TIER_INBOUND extended permit tcp any 10.1.10.0 255.255.255.0 eq 443 log
access-list WEB_TIER_INBOUND extended permit tcp 10.1.100.0 255.255.255.0 10.1.10.0 255.255.255.0 eq 22 log
access-list WEB_TIER_INBOUND extended deny ip any 10.1.10.0 255.255.255.0 log

! Outbound to App Tier
access-list WEB_TO_APP extended permit tcp 10.1.10.0 255.255.255.0 10.1.20.0 255.255.255.0 eq 8080 log
access-list WEB_TO_APP extended permit tcp 10.1.10.0 255.255.255.0 10.1.20.0 255.255.255.0 eq 8443 log
access-list WEB_TO_APP extended permit udp 10.1.10.0 255.255.255.0 10.1.50.0 255.255.255.0 eq 53 log
access-list WEB_TO_APP extended deny ip any any log

! =====================================
! Firewall Configuration - App Tier (DMZ 2)
! =====================================

! Inbound from Web Tier
access-list APP_TIER_INBOUND extended permit tcp 10.1.10.0 255.255.255.0 10.1.20.0 255.255.255.0 eq 8080 log
access-list APP_TIER_INBOUND extended permit tcp 10.1.10.0 255.255.255.0 10.1.20.0 255.255.255.0 eq 8443 log
access-list APP_TIER_INBOUND extended permit tcp 10.1.100.0 255.255.255.0 10.1.20.0 255.255.255.0 eq 22 log
access-list APP_TIER_INBOUND extended deny ip any any log

! Outbound to Data Tier
access-list APP_TO_DATA extended permit tcp 10.1.20.0 255.255.255.0 10.1.30.0 255.255.255.0 eq 3306 log
access-list APP_TO_DATA extended permit tcp 10.1.20.0 255.255.255.0 10.1.30.0 255.255.255.0 eq 5432 log
access-list APP_TO_DATA extended permit tcp 10.1.20.0 255.255.255.0 10.1.30.0 255.255.255.0 eq 1433 log
access-list APP_TO_DATA extended deny ip any any log

! =====================================
! Security Best Practices
! =====================================

! 1. Each tier is separate VLAN
! 2. Explicit firewall rules between tiers
! 3. Default deny all
! 4. Logging enabled for all traffic
! 5. DPI/WAF on Web Tier
! 6. Load balancing for availability
! 7. Encryption between tiers
! 8. No direct internet access from internal
! 9. Regular security audits
! 10. Incident response procedures
